# Regla general. Los rastreadores de IA no necesitan mención aparte para poder entrar
# —"*" ya los cubre— pero sí para lo contrario: quien quiera excluir uno concreto del
# entrenamiento tiene que nombrarlo, porque no obedecen al comodín para eso.
User-agent: *
Allow: /

# El contenido de los cursos se sirve con token de vida corta: indexarlo no tiene sentido
# y solo produciría 403 en el índice.

Disallow: /aprender/
Disallow: /admin/
Disallow: /cuenta/
Disallow: /afiliado
